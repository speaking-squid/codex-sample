from js import document, requestAnimationFrame canvas = document.getElementById("game") ctx = canvas.getContext("2d") # Game settings CELL_SIZE = 20 ROWS = canvas.height // CELL_SIZE COLS = canvas.width // CELL_SIZE direction = 'right' snake = [(5, 5), (4, 5), (3, 5)] food = (10, 10) score = 0 # Key handling def handle_key(evt): global direction key = evt.key if key == 'ArrowUp' and direction != 'down': direction = 'up' elif key == 'ArrowDown' and direction != 'up': direction = 'down' elif key == 'ArrowLeft' and direction != 'right': direction = 'left' elif key == 'ArrowRight' and direction != 'left': direction = 'right' document.addEventListener('keydown', handle_key) # Game functions from random import randint def spawn_food(): global food food = (randint(0, COLS - 1), randint(0, ROWS - 1)) def update(): global score head_x, head_y = snake[0] if direction == 'up': head_y -= 1 elif direction == 'down': head_y += 1 elif direction == 'left': head_x -= 1 elif direction == 'right': head_x += 1 # Wrap around the edges head_x %= COLS head_y %= ROWS new_head = (head_x, head_y) if new_head in snake: reset_game() return snake.insert(0, new_head) if new_head == food: score += 1 document.getElementById("score-value").innerHTML = str(score) spawn_food() else: snake.pop() def reset_game(): global snake, direction, score snake = [(5, 5), (4, 5), (3, 5)] direction = 'right' score = 0 document.getElementById("score-value").innerHTML = '0' spawn_food() def draw(): ctx.fillStyle = 'white' ctx.fillRect(0, 0, canvas.width, canvas.height) # draw food fx, fy = food ctx.fillStyle = 'red' ctx.fillRect(fx * CELL_SIZE, fy * CELL_SIZE, CELL_SIZE, CELL_SIZE) # draw snake ctx.fillStyle = 'green' for x, y in snake: ctx.fillRect(x * CELL_SIZE, y * CELL_SIZE, CELL_SIZE, CELL_SIZE) def game_loop(*args): update() draw() requestAnimationFrame(game_loop) reset_game() requestAnimationFrame(game_loop)
